        function getComputedMetrics() {
            const activeDevices = appState.devices.filter(d => d.state === 'ON' || d.state === 'STANDBY');
            const currentHouseKw = activeDevices.reduce((sum, d) => sum + (d.state === 'ON' ? d.ratedPowerKw : 0.05), 0);
            const todayKwh = appState.devices.reduce((sum, d) => sum + d.todayKwh, 0);
            const monthKwh = appState.devices.reduce((sum, d) => sum + d.monthKwh, 0);
            const estBill = Math.round(monthKwh * appState.tariffRatePerKWh);
            const utilityMeterKwh = appState.utilityMeter.totalConsumedCycleKwh;
            const monitoredKwh = monthKwh;
            const differenceKwh = Math.abs(utilityMeterKwh - monitoredKwh).toFixed(2);
            const coveragePercent = Math.min(100, Math.round((monitoredKwh / utilityMeterKwh) * 100));

            return {
                currentHouseKw: currentHouseKw.toFixed(2),
                todayKwh: todayKwh.toFixed(1),
                monthKwh: monthKwh.toFixed(1),
                estBill,
                utilityMeterKwh,
                monitoredKwh: monitoredKwh.toFixed(1),
                differenceKwh,
                coveragePercent,
                activeCount: activeDevices.filter(d => d.state === 'ON').length,
                totalRooms: appState.rooms.length,
                totalSwitchBoxes: appState.switchBoxes.length,
                totalDevices: appState.devices.length
            };
        }

        function confirmAddSwitchBox() {
            const name = document.getElementById('new-sb-name').value.trim();
            const roomId = document.getElementById('new-sb-room').value;
            const maxLoadKw = parseFloat(document.getElementById('new-sb-load').value);
            const channelCount = parseInt(document.getElementById('new-sb-channels').value);
            if (!name) return showToast('Please enter switch box name');

            const id = 'SB-' + Math.random().toString(36).substr(2, 4).toUpperCase();
            const channels = [];
            for (let i = 1; i <= channelCount; i++) {
                channels.push({ id: `CH-${id}-${i}`, name: `Channel ${i}`, deviceId: null });
            }

            appState.switchBoxes.push({ id, name, roomId, maxLoadKw, voltage: 230, phase: 'Single Phase', status: 'Online', temperature: 36, channels });
            closeModal();
            renderCurrentView();
            showToast(`Switch Box "${name}" created with ${channelCount} channels`);
        }

        function deleteSwitchBox(sbId) {
            appState.switchBoxes = appState.switchBoxes.filter(s => s.id !== sbId);
            renderCurrentView();
            showToast('Switch box deleted successfully');
        }

        function toggleDeviceState(deviceId) {
            const dev = appState.devices.find(d => d.id === deviceId);
            if (!dev) return;
            if (dev.state === 'ON') dev.state = 'OFF';
            else dev.state = 'ON';
            renderCurrentView();
            showToast(`${dev.name} turned ${dev.state}`);
        }

        function applySmartMode(modeName) {
            appState.activeSmartMode = modeName;
            document.getElementById('header-mode-select').value = modeName;
            if (modeName === 'Eco') {
                appState.devices.forEach(d => {
                    if (d.category === 'Climate') { d.state = 'ON'; d.temp = 24; }
                    if (d.category === 'Entertainment') d.state = 'STANDBY';
                });
                showToast('🌱 Eco Mode Activated: AC temperatures optimized & standby curtailed.');
            } else if (modeName === 'Sleep') {
                appState.devices.forEach(d => {
                    if (d.roomId === 'room-living') d.state = 'OFF';
                });
                showToast('🌙 Sleep Mode Activated: Living room shut down, night curves applied.');
            } else if (modeName === 'Away') {
                appState.devices.forEach(d => {
                    if (d.id !== 'dev-kit-ref') d.state = 'OFF';
                });
                showToast('🚪 Away Mode Activated: Non-essential switch boxes disabled.');
            } else {
                appState.devices.forEach(d => { d.state = 'ON'; });
                showToast('⚡ Normal Operation Restored.');
            }
            renderCurrentView();
        }

        function runDemoScenario(scenarioName) {
            if (scenarioName === 'SummerEvening') {
                appState.devices.forEach(d => { d.state = 'ON'; });
                appState.utilityMeter.totalConsumedCycleKwh = 192.4;
                showToast('☀️ Summer Evening Scenario Simulated: All ACs, lights, and appliances active.');
            }
            renderCurrentView();
        }

        function sendCopilotMessage() {
            const input = document.getElementById('copilot-input');
            const query = input.value.trim();
            if (!query) return;
            appendChatMsg('User', query);
            input.value = '';

            setTimeout(() => {
                let reply = "Based on your smart switch box telemetry, your current load is within expected parameters. Shifting heavy appliances to off-peak hours will reduce your monthly bill.";
                if (query.toLowerCase().includes('bill')) reply = "Your estimated bill is ₹" + getComputedMetrics().estBill + ". The primary contributor is the AC units across Bedroom 1 and Living Room.";
                if (query.toLowerCase().includes('room')) reply = "Living Room and Bedroom 1 are currently consuming the highest share of power due to active climate control.";
                if (query.toLowerCase().includes('meter')) reply = "The utility meter reconciliation shows a 95.8% coverage rate with only 0.78 kWh unaccounted for.";
                appendChatMsg('AI', reply);
            }, 600);
        }

        function quickAsk(q) {
            document.getElementById('copilot-input').value = q;
            sendCopilotMessage();
        }

        function appendChatMsg(sender, text) {
            const box = document.getElementById('chat-box');
            box.innerHTML += `
                <div class="flex items-start space-x-3 ${sender === 'User' ? 'justify-end' : ''}">
                    ${sender === 'AI' ? '<div class="w-8 h-8 rounded-xl bg-pink-500/20 text-pink-400 flex items-center justify-center font-bold">AI</div>' : ''}
                    <div class="p-3.5 rounded-2xl ${sender === 'User' ? 'bg-emerald-500/20 text-emerald-200 border border-emerald-500/30' : 'bg-slate-800 text-slate-200'} max-w-md leading-relaxed">
                        ${text}
                    </div>
                </div>
            `;
            box.scrollTop = box.scrollHeight;
        }

            }, 3000);
        }

        function confirmMeterUpdate() {
            const val = parseFloat(document.getElementById('new-meter-reading').value);
            if (isNaN(val)) return showToast('Please enter a valid reading');
            appState.utilityMeter.totalConsumedCycleKwh = val;
            closeModal();
            renderCurrentView();
            showToast('Utility power meter successfully synchronized');
        }
